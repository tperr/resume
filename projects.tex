\section*{Projects}
\textbf{Entropy Emulator}\\
\textit{Objective:} Develop a simple emulator for a computer
\begin{itemize}
    \item Programmed in C++ with a non-complex Makefile.
    \item Reads a file of computer operations (clock tick, memory load, CPU load) and executes commands with assembly-like operations.
    \item Includes operations such as Load/Save Word and Branches.
    \item \url{https://github.com/tperr/Entropy-Emulator}
\end{itemize}

\textbf{Fracture Detector}\\
\textit{Objective:} Use a neural network to detect fractures
\begin{itemize}
    \item Final project for SAT5114: AI in Healthcare.
    \item Implemented in Python using U-Net architecture.
    \item Trained on a dataset provided by \url{figshare.com}.
    \item \url{https://github.com/tperr/Fracture-Detector}
\end{itemize}
